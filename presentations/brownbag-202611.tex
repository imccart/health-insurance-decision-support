\begin{frame}{Takeaways}
  \vspace{0.3cm}
  \begin{enumerate}\setlength{\itemsep}{18pt}
    \item Restrictions on agent commissions should be evaluated together with what replaces agent assistance. Without a replacement, coverage falls. With navigator assistance in its place, coverage is about unchanged and premiums fall.
    \item Changes to how agents are paid have small effects on coverage and household welfare compared with changes in access to assistance.
  \end{enumerate}
\end{frame}

\begin{frame}{Contribution}
  \vspace{0.1cm}
  \begin{itemize}\setlength{\itemsep}{12pt}
    \item<1-> \textbf{Decision assistance and expert advice:} commissioned agents and public navigators compared in the same market\par
      {\footnotesize\color{emoryMid} Schotter (2003) AER; Bettinger et al.\ (2012) QJE; Kling et al.\ (2012) QJE; Finkelstein and Notowidigdo (2019) QJE; Bundorf et al.\ (2024) MS; Egan et al.\ (2019) JPE; Aizawa and Kim (2025) AEJ:Pol; Aizawa et al.\ (WP); Li et al.\ (WP)\par}
    \item<2-> \textbf{Poor choices in health insurance:} assistance reduces dominated plan choices, separately from the steering created by commissions\par
      {\footnotesize\color{emoryMid} Abaluck and Gruber (2011) AER; Sinaiko and Hirth (2011) JHE; Ketcham et al.\ (2012) AER; Bhargava et al.\ (2017) QJE; Liu and Sydnor (2021) AEJ:Pol\par}
    \item<3-> \textbf{Insurer pricing in the exchanges:} decision assistance and agent commissions added to structural models of exchange pricing\par
      {\footnotesize\color{emoryMid} Starc (2014) RAND; Saltzman (2021) RAND; Saltzman (2025) REStat; Tebaldi (2025) REStud\par}
  \end{itemize}
\end{frame}

\begin{frame}[t,label=setting]{Covered California, 2014--2019}
  \vspace{0.3cm}
  \begin{itemize}
    \item California's ACA marketplace: about one million enrolled households a year, 10--12 insurers, 19 rating regions
    \item Four metal tiers with standardized cost sharing 
    \item Premiums vary only by age and region, with subsidies up to 400\% of FPL (set by the second-cheapest silver plan)
    \item Cost-sharing subsidies for households below 250\% of FPL (available only for silver plans)
    \item Average net premium of around \$130 per member per month
  \end{itemize}

    U^{A}_{ijt} = {} &
    \eqterm<2>{\delta_0 + \delta' w_{it}}{coverage}

    \only<5>{\datacard{Insurer and network}

    \only<6>{\datacard{Assistance}

    \only<7>{\datacard{Commission steering}

\begin{frame}[t]{Welfare Measures}
  \vspace{-0.3cm}
  % Consumer surplus
  \only<1-2>{%
    \mrowhead{Consumer Surplus}
    $$CS_i = \frac{1}{\alpha_i}\, \mathbb{E}\Big[\max_j U_{ij}\Big] = \frac{1}{\alpha_i} \ln\Big(\eqterm<1>{e^{V_{i0}}}{uninsured} + \eqterm<1>{e^{V^E_i}}{enrolling}\Big)$$
    \vspace{-0.2cm}
    \begin{itemize}
      \item<1-> Standard consumer surplus, with commission terms set to zero
      \item<2-> Problematic with steering, since the agent channel's weights on premiums and generosity count as household preferences
    \end{itemize}}%
  % Navigator valuation
  \only<3>{%
    \mrowdone{Consumer Surplus}{CS_i = \tfrac{1}{\alpha_i} \ln\big(e^{V_{i0}} + e^{V^E_i}\big)}
    \mrowhead{Consumer Surplus with Navigator Assistance}
    $$V^{nav}_i = \frac{1}{\alpha^{nav}_i} \sum_{j} q_{ijt}\, V^{nav}_{ij}$$
    \vspace{-0.2cm}
    \begin{itemize}
      \item Same choices $q_{ijt}$, valued with a navigator client's utility
      \item Systematic part only, with no agent or commission terms
    \end{itemize}}%
  % Expected costs
  \only<4-5>{%
    \mrowdone{Consumer Surplus}{CS_i = \tfrac{1}{\alpha_i} \ln\big(e^{V_{i0}} + e^{V^E_i}\big)}
    \mrowdone{Consumer Surplus with Navigator Assistance}{V^{nav}_i = \tfrac{1}{\alpha^{nav}_i} \textstyle\sum_j q_{ijt} V^{nav}_{ij}}
    \mrowhead{Expected Costs}
    \vspace{-0.3cm}
    \only<4>{%
      $$V^{obj}_i = \sum_j q_{ijt}\, V^{obj}_{ij}, \qquad V^{obj}_{ij} = -\Big(\eqterm<4>{\text{premium}_{ij}}{premium} + \eqterm<4>{\mathbb{E}[OOP_{ij}]}{expected out-of-pocket} + \eqterm<4>{\tfrac{\rho}{2}\text{Var}[OOP_{ij}]}{variance charge}\Big)$$
      \vspace{-0.3cm}
      \begin{itemize}
        \item Same choices $q_{ijt}$, valued at expected cost in dollars
        \item Out-of-pocket spending from each plan's deductible, coinsurance, and out-of-pocket maximum, applied to 2018 MEPS spending by age and income
      \end{itemize}}%
    \only<5>{%
      $$\text{uninsured:}\quad V^{obj}_{i0} = -\Big(\eqterm<5>{OOP^{u}_{i}}{out-of-pocket} + \eqterm<5>{RP + \Delta m_i\, VSL + D_i}{risk protection, mortality, distress}\Big)$$
      \vspace{-0.3cm}
      \begin{itemize}
        \item The out-of-pocket amount uninsured households actually pay (2018 MEPS), less than their full medical bill
        \item Plus three costs of lacking coverage: lost risk protection, a mortality cost (VSL \$13M), and financial distress above 40\% of income
      \end{itemize}}}%
  % Producer surplus
  \only<6>{%
    \mrowdone{Consumer Surplus}{CS_i = \tfrac{1}{\alpha_i} \ln\big(e^{V_{i0}} + e^{V^E_i}\big)}
    \mrowdone{Consumer Surplus with Navigator Assistance}{V^{nav}_i = \tfrac{1}{\alpha^{nav}_i} \textstyle\sum_j q_{ijt} V^{nav}_{ij}}
    \mrowdone{Expected Costs}{V^{obj}_i = \textstyle\sum_j q_{ijt} V^{obj}_{ij}}
    \mrowhead{Producer Surplus}
    $$PS = \sum_f \Big[\sum_{j \in f} (p_j - mc_j)\, q_j - (1-\beta_f)\, Y_f\Big]$$
    \vspace{-0.2cm}
    \begin{itemize}
      \item Premium revenue less marginal cost (claims net of risk adjustment and reinsurance, plus admin costs)
      \item Less the net cost of commissions
    \end{itemize}}%
  % Government budget
  \only<7>{%
    \mrowdone{Consumer Surplus}{CS_i = \tfrac{1}{\alpha_i} \ln\big(e^{V_{i0}} + e^{V^E_i}\big)}
    \mrowdone{Consumer Surplus with Navigator Assistance}{V^{nav}_i = \tfrac{1}{\alpha^{nav}_i} \textstyle\sum_j q_{ijt} V^{nav}_{ij}}
    \mrowdone{Expected Costs}{V^{obj}_i = \textstyle\sum_j q_{ijt} V^{obj}_{ij}}
    \mrowdone{Producer Surplus}{PS = \textstyle\sum_f \big[\sum_{j \in f} (p_j - mc_j)\, q_j - (1-\beta_f) Y_f\big]}
    \mrowhead{Government Budget}
    $$G = -\big(\text{tax credits} + \text{CSR payments} + \text{uncompensated care} - \text{penalties}\big)$$
    \vspace{-0.2cm}
    \begin{itemize}
      \item Tax credits and cost-sharing reduction payments for enrollees
      \item Uncompensated care: \$2,025 per uninsured person in 2013 (Coughlin et al., 2014), adjusted by year and for the risk of the uninsured pool
    \end{itemize}}%
  % All five
  \only<8>{%
    \mrowdone{Consumer Surplus}{CS_i = \tfrac{1}{\alpha_i} \ln\big(e^{V_{i0}} + e^{V^E_i}\big)}
    \mrowdone{Consumer Surplus with Navigator Assistance}{V^{nav}_i = \tfrac{1}{\alpha^{nav}_i} \textstyle\sum_j q_{ijt} V^{nav}_{ij}}
    \mrowdone{Expected Costs}{V^{obj}_i = \textstyle\sum_j q_{ijt} V^{obj}_{ij}}
    \mrowdone{Producer Surplus}{PS = \textstyle\sum_f \big[\sum_{j \in f} (p_j - mc_j)\, q_j - (1-\beta_f) Y_f\big]}
    \mrowdone{Government Budget}{G = -(\text{tax credits} + \text{CSR} + \text{uncomp.\ care} - \text{penalties})}}
\end{frame}

\section{Conclusion}
\sectionframe

\begin{frame}{Conclusion}
  \begin{itemize}\setlength{\itemsep}{10pt}
    \item<1-> Decision assistance reduces dominated plan choices by 42\%, and agent-assisted households respond to a \$1 commission as they would to a \$\commPremRatio{} reduction in net premium.
    \item<2-> Banning commissions raises the uninsured share by 6.1pp if agent clients receive no other assistance. If navigators replace agents, the uninsured share is about unchanged and premiums fall.
    \item<3-> Changes to the commission schedule change the uninsured share by at most 0.7pp. Aligning commissions with household value is the only change that raises all household welfare measures.
    \item<4-> Expanding navigator assistance raises household welfare and lowers subsidy spending, not counting the cost of the additional navigators.
  \end{itemize}
\end{frame}

\begin{frame}{Final Takeaways}
  \vspace{0.3cm}
  \begin{enumerate}\setlength{\itemsep}{18pt}
    \item<1-> Restrictions on agent commissions should be evaluated together with what replaces agent assistance. Without a replacement, coverage falls. With navigator assistance in its place, coverage is about unchanged and premiums fall.
    \item<2-> Changes to how agents are paid have small effects on coverage and household welfare compared with changes in access to assistance.
  \end{enumerate}
\end{frame}

  \begin{itemize}\setlength{\itemsep}{6pt}

\begin{frame}[t,label=premiums]{Premiums and subsidies}
  \vspace{-0.4cm}
  \large
  $$
  p_{ijt} = \max\Big\{
  \eqterm<2>{\sigma_{it}\, p_{jmt}}{full premium}
  \;-\; \eqterm<3>{\max\{\sigma_{it}\, p_{bmt} - \zeta_{it},\, 0\}}{subsidy}
  ,\; \eqterm<4>{0}{floor}
  \Big\}
  $$
  \normalsize
  \vspace{-0.4cm}
  \begin{overlayarea}{\textwidth}{0.5\textheight}
  \only<1>{%
    \vspace{0.3cm}
    {The net premium household $i$ pays for plan $j$ in rating region $m$, year $t$}}
  \only<2>{%
    \begin{itemize}
      \item $p_{jmt}$ is the plan's base premium in the household's rating region
      \item $\sigma_{it}$ is the household's age rating factor (a 64-year-old pays up to three times what a 21-year-old pays)
    \end{itemize}}
  \only<3>{%
    \begin{itemize}
      \item Available for incomes of 100--400\% of FPL
      \item $p_{bmt}$ is the base premium of the benchmark plan, the second-cheapest silver plan in the region
      \item $\zeta_{it}$ is the household's contribution cap, from 2\% of income at 100\% of FPL to 9.5\% at 400\% (2014)
      \item The same dollar subsidy applies to any plan
    \end{itemize}}
  \only<4>{%
    \begin{itemize}
      \item When the subsidy exceeds a plan's full premium, the household pays nothing
      \item Some bronze plans are free to low-income households
    \end{itemize}}
  \only<5>{%
    \begin{itemize}
      \item Raising the benchmark premium raises every household's subsidy
      \item The benchmark's own net price is unchanged for subsidized households
      \item Every rival plan's net price falls
    \end{itemize}}
  \end{overlayarea}

  \cornerbutton{setting}{back to setting}
\end{frame}

\begin{frame}[t,label=ident]{Identification of key parameters}
  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Premium:} variation across plans, the mandate penalty phasing in (2014--2016) and ending (2019), and kinks in the subsidy formula that make some plans free
    \item \textbf{Assistance terms:} assisted and unassisted households selecting different plans from the same menu within a region and year
    \item \textbf{Commission terms:} commissions differ across insurers at a given actuarial value and change over time within some insurers
    \item \textbf{Insurer effects:} brand indicators absorb networks and other unobserved features, so commission effects come from within-insurer variation
  \end{itemize}

\begin{frame}[t,label=openq]{Open questions}
  \vspace{-0.2cm}
  \footnotesize
  \onslide<1->{{\bfseries\color{emoryBlue}Do households choose their assistance channel?}
  \begin{itemize}\setlength{\itemsep}{0pt}
    \item \textbf{Data limitation:} we only observe the channel for enrolled households, so we cannot estimate a model of channel choice for uninsured households
    \item A three-level nest allowing for channel choice assumes that higher commissions also raise agent use
    \item Across regions, agent use is not strongly correlated with commissions or with number of plans available, so it's not clear that households are making much of a ``choice'' here
  \end{itemize}}

  \vspace{0cm}
  \onslide<2->{{\bfseries\color{emoryBlue}How do insurers set commissions?}
  \begin{itemize}\setlength{\itemsep}{0pt}
    \item We assume each carrier sets commissions to maximize profit, weighing the profit from steered enrollment against the net cost of the commission
    \item The commission this implies is very sensitive to estimated margins, since a \$10 change in margins changes it by about as much as the average commission (\$13)
    \item So the condition fits observed commissions only with carrier constants $\delta_f$, which absorb much of the variation across carriers 
    \item ...but most commission variation is across carriers and constant over time anyway
  \end{itemize}}
